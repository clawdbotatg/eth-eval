pragma solidity ^0.8.20;

interface IERC20 {
    function balanceOf(address) external view returns (uint256);
    function transfer(address, uint256) external returns (bool);
    function transferFrom(address, address, uint256) external returns (bool);
}

/// ERC-4626-style: yield arrives as a direct transfer and is shared pro-rata.
/// The inflation attack is stopped by a virtual share/asset offset, NOT by
/// ignoring donations - ignoring them would throw the yield away.
contract Vault {
    IERC20 public immutable asset;
    uint256 public totalShares;
    mapping(address => uint256) public shares;

    uint256 private constant OFFSET = 1e6;   // virtual shares

    constructor(address _asset) { asset = IERC20(_asset); }

    function totalAssets() public view returns (uint256) {
        return asset.balanceOf(address(this));
    }

    function deposit(uint256 assets) external returns (uint256 s) {
        s = (assets * (totalShares + OFFSET)) / (totalAssets() + 1);   // floor
        require(s > 0, "zero shares");
        shares[msg.sender] += s;
        totalShares += s;
        require(asset.transferFrom(msg.sender, address(this), assets), "pull failed");
    }

    function redeem(uint256 s) external returns (uint256 assets) {
        assets = (s * (totalAssets() + 1)) / (totalShares + OFFSET);   // floor
        shares[msg.sender] -= s;
        totalShares -= s;
        require(asset.transfer(msg.sender, assets), "transfer failed");
    }
}
