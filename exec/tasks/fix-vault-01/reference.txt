pragma solidity ^0.8.20;

interface IERC20 {
    function balanceOf(address) external view returns (uint256);
    function transfer(address, uint256) external returns (bool);
    function transferFrom(address, address, uint256) external returns (bool);
}

contract Vault {
    IERC20 public immutable asset;
    uint256 public totalShares;
    uint256 public totalAssets;          // internal accounting: donations do not count
    mapping(address => uint256) public shares;

    uint256 private constant OFFSET = 1000;   // virtual shares/assets

    constructor(address _asset) { asset = IERC20(_asset); }

    function deposit(uint256 amount) external returns (uint256 s) {
        s = (amount * (totalShares + OFFSET)) / (totalAssets + 1);
        require(s > 0, "zero shares");
        shares[msg.sender] += s;
        totalShares += s;
        totalAssets += amount;
        require(asset.transferFrom(msg.sender, address(this), amount), "pull failed");
    }

    function withdraw(uint256 s) external returns (uint256 amount) {
        amount = (s * (totalAssets + 1)) / (totalShares + OFFSET);
        shares[msg.sender] -= s;
        totalShares -= s;
        totalAssets -= amount;
        require(asset.transfer(msg.sender, amount), "transfer failed");
    }
}
